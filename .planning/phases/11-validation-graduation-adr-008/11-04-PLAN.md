**Step 2 — REVISED 2026-05-05 per reviewer MEDIUM #10: suspend-then-uninstall-then-resume sequence:**

```bash
if kubectl --context=k3d-spoke-capsule cluster-info >/dev/null 2>&1 \
   && kubectl --context=k3d-hub-flux cluster-info >/dev/null 2>&1; then
  # Pre-condition check: only run uninstall if helm-managed capsule-proxy actually exists
  if ! command -v helm >/dev/null 2>&1; then
    echo "WARN: helm not installed; skipping uninstall step (CI may not have helm)"
  elif ! helm --kube-context=k3d-spoke-capsule list -n capsule-system 2>/dev/null | grep -q '^capsule-proxy[[:space:]]'; then
    echo "INFO: no helm-installed capsule-proxy found; skipping uninstall (Flux already canonical)"
  else
    # REVISED 2026-05-05 per reviewer MEDIUM #10:
    # Step 2a: suspend the relevant Kustomizations FIRST so Flux cannot re-install the chart
    # in an inconsistent state mid-uninstall.
    flux suspend kustomization poc-capsule -n flux-system 2>/dev/null || true
    flux suspend kustomization poc-capsule-spoke -n flux-system 2>/dev/null || true
    echo "suspended: poc-capsule + poc-capsule-spoke (pre-uninstall race mitigation per reviewer MEDIUM #10)"

    # Step 2b: uninstall the imperative helm release
    helm --kube-context=k3d-spoke-capsule uninstall capsule-proxy -n capsule-system || true
    echo "removed: helm install artifact for capsule-proxy"

    # Step 2c: REVISED 2026-05-05 -- drain wait for chart resources to actually disappear.
    # Without this, the resume could re-install before the previous resources are fully gone.
    for i in 1 2 3; do
      if ! kubectl --context=k3d-spoke-capsule get deploy capsule-proxy -n capsule-system >/dev/null 2>&1; then
        echo "drain complete: capsule-proxy Deployment gone after $((i*10))s"
        break
      fi
      sleep 10
    done

    # Step 2d: resume Kustomizations so the post-push Flux reconcile is the canonical install path
    flux resume kustomization poc-capsule-spoke -n flux-system 2>/dev/null || true
    flux resume kustomization poc-capsule -n flux-system 2>/dev/null || true
    echo "resumed: poc-capsule + poc-capsule-spoke (post-uninstall; ready for push event canonical reconcile)"
  fi
fi
```

This removes the imperative artifact AND prevents the race with Flux re-installing the chart in an inconsistent state.
