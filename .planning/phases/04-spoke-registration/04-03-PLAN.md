**2. `wait_for_token_secret <spoke>`** — D-11 bounded retry (10×1s). RESEARCH.md Example 2 lines 605-638. Sets globals `TOKEN_B64`, `CA_B64`, `TOKEN_DECODED`. Fails with `kubectl describe` remediation:

```bash
wait_for_token_secret() {
  local spoke="$1"
  local timeout_s=${TOKEN_WAIT_SECONDS}
  local token="" ca=""
  local i
  for i in $(seq 1 "${timeout_s}"); do
    token="$(kubectl --context "k3d-${spoke}" -n flux-system \
      get secret flux-reconciler-token -o jsonpath='{.data.token}' 2>/dev/null || true)"
    ca="$(kubectl --context "k3d-${spoke}" -n flux-system \
      get secret flux-reconciler-token -o jsonpath='{.data.ca\.crt}' 2>/dev/null || true)"
    [[ -n "${token}" && -n "${ca}" ]] && break
    sleep 1
  done
  if [[ -z "${token}" || -z "${ca}" ]]; then
    fail "flux-reconciler-token not populated on ${spoke} after ${timeout_s}s.
       Inspect: kubectl --context k3d-${spoke} -n flux-system describe secret/flux-reconciler-token
       Possible: SA missing (orphan Secret); apiserver under load; controller race.
       Fix: rerun: bash scripts/register-spokes-for-flux.sh"
    exit 1
  fi
  TOKEN_B64="${token}"
  CA_B64="${ca}"
  TOKEN_DECODED="$(echo "${token}" | base64 -d)"
}
```

verify_spoke_credential_layer() {
  local spoke="$1" token="$2"
  local server="https://k3d-${spoke}-server-0:6443"
  local expected_node="k3d-${spoke}-server-0"

  # Gate 1: Presence
  local secret_data
  secret_data="$(kubectl --context "${HUB_CTX}" -n "${FLUX_NS}" \
    get secret "${spoke}-kubeconfig" -o jsonpath='{.data.value\.yaml}' 2>/dev/null || true)"

       Fix: rerun: bash scripts/register-spokes-for-flux.sh"
    exit 1
  fi

  # Gate 2: Decode + parse
  local decoded_server
  decoded_server="$(echo "${secret_data}" | base64 -d | yq eval '.clusters[0].cluster.server' - 2>/dev/null || true)"
  if [[ "${decoded_server}" != "${server}" ]]; then
