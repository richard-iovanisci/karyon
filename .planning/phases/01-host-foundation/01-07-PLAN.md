**3. Create `tests/bats/preflight-pre02.bats`** (Ubuntu 24.04 distro detection):

Tests the /etc/os-release parsing logic. Uses a temporary file to simulate different distros.

```bash
#!/usr/bin/env bats
# tests/bats/preflight-pre02.bats
# Unit tests for PRE-02: Ubuntu 24.04 distro detection via /etc/os-release parsing.

load '../test_helper'

setup() {
  TMPDIR="$(mktemp -d)"
}

teardown() {
  rm -rf "$TMPDIR"
}

@test "PRE-02: Ubuntu 24.04 detected as target" {
  cat > "${TMPDIR}/os-release" <<'EOF'
ID=ubuntu
VERSION_ID="24.04"
PRETTY_NAME="Ubuntu 24.04.1 LTS"
EOF
  # shellcheck source=/dev/null
  . "${TMPDIR}/os-release"
  [ "${ID:-}" = "ubuntu" ]
  [ "${VERSION_ID:-}" = "24.04" ]
}

@test "PRE-02: Ubuntu 22.04 is not the target version" {
  cat > "${TMPDIR}/os-release" <<'EOF'
ID=ubuntu
VERSION_ID="22.04"
PRETTY_NAME="Ubuntu 22.04.5 LTS"
EOF
  # shellcheck source=/dev/null
  . "${TMPDIR}/os-release"
  [ "${VERSION_ID:-}" != "24.04" ]
}

@test "PRE-02: non-Ubuntu distro ID detected" {
  cat > "${TMPDIR}/os-release" <<'EOF'
ID=debian
VERSION_ID="12"
PRETTY_NAME="Debian GNU/Linux 12 (bookworm)"
EOF
  # shellcheck source=/dev/null
  . "${TMPDIR}/os-release"
  [ "${ID:-}" != "ubuntu" ]
}

@test "PRE-02: VERSION_ID 24.04 matches expected" {
  local version="24.04"
  case "$version" in
    24.04) result="target" ;;
    22.04) result="warn" ;;
    *)     result="warn" ;;
  esac
  [ "$result" = "target" ]
}
```

```bash
  TMPDIR="$(mktemp -d)"
  ENV_FILE="${TMPDIR}/.env"

teardown() {
  rm -rf "$TMPDIR"
}

@test "PRE-09: .env file missing returns non-zero for test -f" {
  run test -f "${TMPDIR}/missing.env"
  [ "$status" -ne 0 ]
}

  touch "$ENV_FILE"
  run test -f "$ENV_FILE"
  [ "$status" -eq 0 ]
}

  echo 'GITHUB_OWNER=myuser' > "$ENV_FILE"
  val="$(grep -E "^GITHUB_OWNER=" "$ENV_FILE" | cut -d= -f2-)"
  [ -n "$val" ]
}

@test "PRE-09: GITHUB_TOKEN placeholder value is non-empty" {
  echo 'GITHUB_TOKEN=ghp_REPLACE_WITH_PAT_WITH_repo_SCOPE' > "$ENV_FILE"

  [ -n "$val" ]
}

@test "PRE-09: missing key returns empty value" {
```

```bash
#!/usr/bin/env bats
# tests/bats/preflight-pre11.bats
# Unit tests for PRE-11: systemd-resolved 127.0.0.53 stub detection.

load '../test_helper'

setup() {
  TMPDIR="$(mktemp -d)"
  RESOLV_FILE="${TMPDIR}/resolv.conf"
}

teardown() {
  rm -rf "$TMPDIR"
}

@test "PRE-11: 127.0.0.53 nameserver detected" {
  echo 'nameserver 127.0.0.53' > "$RESOLV_FILE"
  run grep -qE '^nameserver[[:space:]]+127\.0\.0\.53' "$RESOLV_FILE"
  [ "$status" -eq 0 ]
}

@test "PRE-11: real nameserver not flagged as stub" {
  echo 'nameserver 1.1.1.1' > "$RESOLV_FILE"
  run grep -qE '^nameserver[[:space:]]+127\.0\.0\.53' "$RESOLV_FILE"
  [ "$status" -ne 0 ]
}

@test "PRE-11: 127.0.0.1 is not flagged (different loopback)" {
  echo 'nameserver 127.0.0.1' > "$RESOLV_FILE"
  run grep -qE '^nameserver[[:space:]]+127\.0\.0\.53' "$RESOLV_FILE"
  [ "$status" -ne 0 ]
}

@test "PRE-11: comment line with 127.0.0.53 not matched" {
  echo '# nameserver 127.0.0.53' > "$RESOLV_FILE"
  run grep -qE '^nameserver[[:space:]]+127\.0\.0\.53' "$RESOLV_FILE"
  [ "$status" -ne 0 ]
}

@test "PRE-11: empty resolv.conf not matched" {
  touch "$RESOLV_FILE"
  run grep -qE '^nameserver[[:space:]]+127\.0\.0\.53' "$RESOLV_FILE"
  [ "$status" -ne 0 ]
}
```
