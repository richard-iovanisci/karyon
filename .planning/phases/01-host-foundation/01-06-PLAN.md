**PRE-09: .env secrets presence** (D-10 — presence + non-empty per key only; no GitHub API):
```bash
# ---------- 9. .env secrets (PRE-09) ----------
section ".env secrets presence (PRE-09)"
ENV_FILE="${REPO_ROOT}/.env"
if [[ ! -f "$ENV_FILE" ]]; then
  fail ".env not found — run: cp .env.example .env && edit it (fill in GITHUB_OWNER, GITHUB_REPO, GITHUB_TOKEN)"
else
  pass ".env file exists"
  for key in GITHUB_OWNER GITHUB_REPO GITHUB_TOKEN; do
    val="$(grep -E "^${key}=" "$ENV_FILE" 2>/dev/null | cut -d= -f2-)"
    if [[ -z "$val" ]]; then
      fail "${key} is missing or empty in .env — edit .env and set a non-empty value"
    else
      pass "${key} is set (value hidden)"
    fi
  done
fi
```

**PRE-12: clock skew > 30s** (RESEARCH.md §Code Examples §PRE-12):
```bash
# ---------- 12. Clock skew check (PRE-12) ----------
section "Clock skew check (PRE-12)"
sys_epoch="$(date +%s 2>/dev/null)"
hw_epoch="$(sudo hwclock --get --utc 2>/dev/null | awk '{print $1, $2}' | date -f - +%s 2>/dev/null || echo "")"
if [[ -z "$sys_epoch" || -z "$hw_epoch" ]]; then
  warn "could not determine clock skew (hwclock not accessible or date parsing failed)"
else
  skew=$(( sys_epoch - hw_epoch ))
  skew_abs="${skew#-}"  # absolute value via string manipulation
  if (( skew_abs > 30 )); then
    fail "clock skew ${skew_abs}s exceeds 30s threshold.
      Fix: sudo hwclock -s  (or: systemctl start hwclock-resume.service)"
  else
    pass "clock skew ${skew_abs}s (within 30s threshold)"
  fi
fi
```

**PRE-13: cgroup v2 purity** (RESEARCH.md §Ref-7):
```bash
# ---------- 13. cgroup v2 purity (PRE-13) ----------
section "cgroup v2 purity (PRE-13)"
cg_type="$(stat -fc %T /sys/fs/cgroup 2>/dev/null)"
if [[ "$cg_type" == "cgroup2fs" ]]; then
  pass "cgroup v2 pure (cgroup2fs)"
elif [[ "$cg_type" == "tmpfs" ]]; then
  fail "cgroup v1 or hybrid detected (got: tmpfs) — k3d + GPU workloads require pure cgroup v2.
      Fix: check %USERPROFILE%\\.wslconfig for kernelCommandLine overrides; ensure WSL kernel ≥ 5.15"
else
  warn "unable to determine cgroup type (got: ${cg_type:-empty}) — may be harmless on this kernel"
fi
```

**PRE-14: bridge-curl probe** (RESEARCH.md §Pitfall 7 — D-04 runs regardless):
```bash
# ---------- 14. Bridge networking probe (PRE-14) ----------
section "Bridge networking probe (PRE-14)"
if ! have docker || ! docker info >/dev/null 2>&1; then
  warn "docker not reachable — skipping bridge probe (install docker first)"
else
  # Ensure k8s-net exists for the probe; create transiently if absent
  net_created=0
  if ! docker network ls --format '{{.Name}}' 2>/dev/null | grep -qx "k8s-net"; then
    docker network create k8s-net >/dev/null 2>&1 || true
    net_created=1
  fi

  probe_pass=0
  A_ID=$(docker run -d --rm --network k8s-net --name preflight-probe-a \
    busybox sh -c 'echo ok > /tmp/h && httpd -f -p 8080 -h /tmp' 2>/dev/null || echo "")
  if [[ -n "$A_ID" ]]; then
    result=$(docker run --rm --network k8s-net busybox \
      wget -qO- http://preflight-probe-a:8080/h 2>/dev/null || echo "FAIL")
    docker rm -f "$A_ID" >/dev/null 2>&1 || true
    [[ "$result" == "ok" ]] && probe_pass=1
  fi

  # Remove k8s-net only if we created it (leave it if Phase 2 already created it)
  (( net_created == 1 )) && docker network rm k8s-net >/dev/null 2>&1 || true

  if (( probe_pass == 1 )); then
    pass "bridge-curl probe: container DNS resolution on k8s-net works"
  else
    fail "bridge-curl probe FAILED — Docker custom network DNS is broken.
      In mirrored mode this is a known limitation (moby/moby#48201).
      Fix: set networkingMode=NAT in %USERPROFILE%\\.wslconfig, then run 'wsl --shutdown'.
      See docs/wsl-networking.md for the migration procedure."
  fi
fi
```
