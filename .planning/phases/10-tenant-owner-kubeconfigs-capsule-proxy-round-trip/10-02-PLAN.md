**STEP 2 — Stage capsule-proxy on spoke-capsule (only if absent):**

```bash
if [[ "$PROXY_PRESTATE" == "absent" ]]; then
  # Plan 10-02 staging: render the kustomize directory + apply directly to spoke-capsule.
  # This is the suspend+apply pattern from Phase 9 Plan 09-04, adapted for Phase 10:
  # - We do NOT modify any files
  # - We do NOT push to origin (D-08-13 push-gate stays deferred)
  # - We DO reach into spoke-capsule with --context=k3d-spoke-capsule to render+apply locally

  # The helmrelease.yaml is a Flux CRD; spoke-capsule has NO Flux CRDs (ADR-004 invariant).
  # So we cannot directly `kubectl apply -k pocs/capsule/proxy/` to spoke (the HelmRelease kind isn't recognized).
  # Instead, render the chart to plain Kubernetes manifests via helm template, then apply those.

  # 2a: Read chart pin from OCIRepository
  CHART_VERSION=$(yq eval '.spec.reference.tag' pocs/capsule/proxy/ocirepository.yaml)
  CHART_URL=$(yq eval '.spec.url' pocs/capsule/proxy/ocirepository.yaml)
  # Expected: CHART_VERSION=0.12.0, CHART_URL=oci://ghcr.io/projectcapsule/charts/capsule-proxy

  # 2b: Read values from helmrelease.yaml (use Flux's values: stanza)
  yq eval '.spec.values' pocs/capsule/proxy/helmrelease.yaml > /tmp/proxy-values-$$.yaml

  # 2c: Render chart to plain manifests
  helm template capsule-proxy "${CHART_URL}" \
    --version "${CHART_VERSION}" \
    --namespace capsule-system \
    --values /tmp/proxy-values-$$.yaml \
    > /tmp/proxy-rendered-$$.yaml

  # 2d: Apply rendered manifests to spoke-capsule
  kubectl --context=k3d-spoke-capsule apply -f /tmp/proxy-rendered-$$.yaml

  # 2e: Wait for proxy Pod to be Ready (max 90s — chart certgen Job + proxy Pod startup)
  kubectl --context=k3d-spoke-capsule -n capsule-system wait --for=condition=Available deploy/capsule-proxy --timeout=90s

  # 2f: Cleanup tmpfiles
  rm -f /tmp/proxy-values-$$.yaml /tmp/proxy-rendered-$$.yaml
fi
```
