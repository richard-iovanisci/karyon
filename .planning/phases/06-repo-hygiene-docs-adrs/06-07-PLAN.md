Rationale recap:
- `default: true` — enable the full markdownlint rule set as a baseline.
- `MD013: false` — disable line-length. This repo's CONTEXT.md / PLAN files use natural-prose long lines; enforcing line-length wraps adds noise without value.
- `MD033 allowed_elements: ["br", "div"]` — allow `<br/>` for inline newlines in Mermaid table cells (the topology diagram uses them) and `<div>` for any future structured callout.
- `MD041: true` — keep first-line-h1 enforced for all top-level docs. This is what the architecture.md MD041-safety check (Plan 03 Task 1) depends on; if a doc accidentally starts with a Mermaid block instead of an h1, CI catches it.

```yaml
# .github/workflows/ci.yml
# karyon CI — 5 parallel jobs (D-09 + D-11 + D-12).
# Triggers: every push to any branch + every pull_request.
# Branch protection on `main` requires all 5 to be green to merge (configured
# manually in GitHub UI; documented in README.md "Repo Hygiene & CI").

name: CI
on:
  push:
    branches: ['**']
  pull_request:

permissions:
  contents: read

jobs:
  shellcheck:
    name: shellcheck
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@<ACTIONS_CHECKOUT_SHA>  # v6.0.2
      - uses: ludeeus/action-shellcheck@<ACTION_SHELLCHECK_SHA>  # v2.0.0
        with:
          scandir: './scripts'
          severity: warning

  markdownlint:
    name: markdownlint
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@<ACTIONS_CHECKOUT_SHA>  # v6.0.2
      - uses: DavidAnson/markdownlint-cli2-action@<MARKDOWNLINT_ACTION_SHA>  # v23
        with:
          globs: '**/*.md'
          # .markdownlint.json is auto-discovered at the repo root.

  kubeconform:
    name: kubeconform
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@<ACTIONS_CHECKOUT_SHA>  # v6.0.2
      - name: Install kubeconform v0.7.0
        run: |
          curl -fsSL https://github.com/yannh/kubeconform/releases/download/v0.7.0/kubeconform-linux-amd64.tar.gz \
            | sudo tar -xz -C /usr/local/bin kubeconform
          sudo chmod +x /usr/local/bin/kubeconform
      - name: Validate clusters/ and examples/
        env:
          # Pinned SHA for the datreeio/CRDs-catalog (D-10 + RESEARCH Open Question #2).
          # Bump this SHA when Flux CRDs change in upstream + a re-validation run is required.
          DATREE_SHA: <DATREE_SHA>
        run: |
          find clusters examples -name '*.yaml' -not -name 'kustomization.yaml' -print0 \
            | xargs -0 kubeconform \
                -strict \
                -ignore-missing-schemas \
                -skip CustomResourceDefinition,Kustomization \
                -schema-location default \
                -schema-location "https://raw.githubusercontent.com/datreeio/CRDs-catalog/${DATREE_SHA}/{{.Group}}/{{.ResourceKind}}_{{.ResourceAPIVersion}}.json" \
                -summary

  prune-lint-bats:
    name: prune-lint-bats
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@<ACTIONS_CHECKOUT_SHA>  # v6.0.2
      - name: Install bats from apt
        run: |
          sudo apt-get update
          sudo apt-get install -y bats
      - name: Run prune-lint test (DESTROY-04 — D-12 reuses existing bats test verbatim)
        run: bats tests/bats/destroy-rebuild-01-static.bats

  gitleaks-scan:
    name: gitleaks-scan
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@<ACTIONS_CHECKOUT_SHA>  # v6.0.2
        with:
          # CRITICAL (Pitfall 3): full history needed for `gitleaks git --log-opts="--all"`.
          # Default is fetch-depth: 1 (shallow); gitleaks would silently false-pass on
          # historical leaks.
          fetch-depth: 0
      - name: Install gitleaks v8.30.1
        run: |
          curl -fsSL https://github.com/gitleaks/gitleaks/releases/download/v8.30.1/gitleaks_8.30.1_linux_x64.tar.gz \
            | sudo tar -xz -C /usr/local/bin gitleaks
          sudo chmod +x /usr/local/bin/gitleaks
      - name: Scan full repo + history
        run: |
          gitleaks git \
            --verbose \
            --redact \
            --log-opts="--all"
```
