<!-- Flux v2 HelmRelease v2 API contract (verified 2026-04-29 against fluxcd.io/flux/components/helm/api/v2/) -->
HelmRelease v2:
  apiVersion: helm.toolkit.fluxcd.io/v2
  kind: HelmRelease
  metadata.{name,namespace}: required
  spec.interval: required (Go duration string, e.g. "5m")
  spec.releaseName: optional (defaults to metadata.name)
  spec.targetNamespace: optional (where chart resources land; "capsule-system" per Claude's Discretion)
  spec.storageNamespace: optional (where Helm release Secret lives; "capsule-system" co-located)
  spec.chartRef: required (post-2024 modern shape)
    spec.chartRef.kind: "OCIRepository" | "HelmChart"
    spec.chartRef.name: <local OCIRepository name>
    spec.chartRef.namespace: optional (defaults to HelmRelease namespace)
  spec.kubeConfig: optional (cross-cluster install via remote kubeconfig Secret)
    spec.kubeConfig.secretRef.name: <Secret name in HelmRelease namespace>
    spec.kubeConfig.secretRef.key: required (P18 / P40 — NEVER omit)
  spec.install.createNamespace: optional bool (default false)
  spec.install.crds: "Skip" | "Create" | "CreateReplace" (Capsule chart contract = CreateReplace)
  spec.install.remediation.retries: int (default 0)
  spec.upgrade.crds: same shape as install.crds
  spec.upgrade.remediation.retries: int
  spec.values: object (chart values)
  spec.dependsOn: list (NOT used on operator HR — proxy depends on operator, not vice versa)
