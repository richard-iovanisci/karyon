```yaml
apiVersion: capsule.clastix.io/v1beta2

  name: hello-world-seeded
  labels:
    karyon.io/poc: capsule

  tenantSelector:
    matchLabels: {}                 # match-all (canonical k8s pattern per A1)

        labels:
          karyon.io/managed-by: capsule-global-tenant-resource

        - apiVersion: v1
          kind: Pod
          metadata:
            name: hello-world
            labels:
              app: hello-world
          spec:
            containers:
              - name: nginx
                image: docker.io/library/nginx:alpine    # FQCI required (Pitfall 13-P1)

                  - containerPort: 80
                    name: http

          kind: Service
          metadata:
            name: hello-world
            labels:
              app: hello-world
          spec:
            type: ClusterIP
            selector:
              app: hello-world
            ports:
              - port: 80
                targetPort: http
                protocol: TCP
```
